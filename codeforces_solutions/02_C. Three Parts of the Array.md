> Another Approach (Binary Search-Not Optimal)
```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<long long> a(n);

    long long total = 0;

    for(auto &x : a) {
        cin >> x;
        total += x;
    }

    vector<long long> suffix(n);

    suffix[n-1] = a[n-1];

    for(int i = n-2; i >= 0; i--) {
        suffix[i] = suffix[i+1] + a[i];
    }

    long long prefix = 0;
    long long ans = 0;

    for(int i = 0; i < n-1; i++) {
        prefix += a[i];

        long long target = total - 2 * prefix;

        auto it = lower_bound(suffix.begin() + i + 1,
                              suffix.end(), target,
                              greater<long long>());

        if(it != suffix.end() && *it == target) {
            ans = max(ans, prefix);
        }
    }

    cout << ans << '\n';

    return 0;
}
```
